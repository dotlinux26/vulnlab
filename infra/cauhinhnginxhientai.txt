# 2. HTTPS SERVICES (Sếp Juna's Legacy)
# ====================================================

# VULN SUBDOMAIN - HỆ THỐNG LAB CHÍNH
server {
    listen 443 ssl;
    server_name vuln.ghedahaui.online;

    ssl_certificate /etc/letsencrypt/live/vuln.ghedahaui.online/fullchain.pem;
    ssl_certificate_key /etc/letsencrypt/live/vuln.ghedahaui.online/privkey.pem;
    client_max_body_size 50M;
 
    location /uploads {
            # Bypass authentication middleware
            auth_request off;
            
            # Proxy tới backend
            proxy_pass http://localhost:6667;
            proxy_set_header Host $host;
            proxy_set_header X-Real-IP $remote_addr;
            proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
            proxy_set_header X-Forwarded-Proto $scheme;
            
            # CORS headers
            add_header 'Access-Control-Allow-Origin' '*' always;
            add_header 'Access-Control-Allow-Methods' 'GET, OPTIONS' always;
            
            # Cache
            add_header Cache-Control "public, max-age=3600" always;
        }

    # --- THÊM KHỐI NÀY VÀO TRƯỚC location / ---
    # Ép các yêu cầu vào /labs phải sang Backend (6667) chứ không được sang Frontend (5173)
    location /labs {
        proxy_pass http://localhost:6667;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        add_header Cache-Control "no-store, no-cache, must-revalidate";
    }


    # [HỆ THỐNG LAB DOCKER MỚI] - Đẩy giao thông sang Gateway ở port 7777
    location /labs-env/ {
        proxy_pass http://localhost:7777/; 
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        
        # Hỗ trợ WebSockets (Nếu sau này làm lab cần XSS realtime)
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection "upgrade";
    }
        

    # 1. Giao diện chính (Frontend) - Đưa ra ngay trang chủ /
    location / {
        proxy_pass http://localhost:5173; 
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection 'upgrade';
        proxy_set_header Host $host;
        proxy_cache_bypass $http_upgrade;
        add_header Cross-Origin-Opener-Policy "unsafe-none";
        add_header Cross-Origin-Opener-Policy "same-origin-allow-popups";
    }

    location /api/payment {
            proxy_pass http://127.0.0.1:6668;
            proxy_set_header Host $host;
            proxy_set_header X-Real-IP $remote_addr;
            proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
            proxy_set_header X-Forwarded-Proto $scheme;
        }

    # 2. Hệ thống API Backend
    location /api {
        proxy_pass http://localhost:6667; # Lưu ý: Nếu trong code backend đã có tiền tố /api thì để nguyên, nếu chưa thì dùng /api/
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }

    # ==============================================================
    # 🎯 THÊM KHỐI NÀY CHO CHAT SERVER (WEBSOCKET PORT 6668)
    # ==============================================================
    location /socket.io/ {
        proxy_pass http://127.0.0.1:6668;
        
        # Cấu hình Upgrade cần thiết cho WebSocket
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection "Upgrade";
        proxy_set_header Host $host;
        
        # Headers bảo vệ và nhận diện IP thật
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;

        # Kéo dài thời gian timeout để không bị ngắt kết nối chat
        proxy_read_timeout 86400;
        proxy_send_timeout 86400;
    }
 
}
